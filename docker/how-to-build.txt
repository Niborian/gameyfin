Local replacement image:

Windows:
  .\docker\build-image.ps1 -ImageName gameyfin -ImageTag variant-local

Linux/macOS:
  ./docker/build-image.sh --image gameyfin --tag variant-local

Production mode requires GAMEYFIN_KEYSTORE_PASSWORD because plugin JARs are signed:
  .\docker\build-image.ps1 -ImageName gameyfin -ImageTag variant-local -Production
  ./docker/build-image.sh --image gameyfin --tag variant-local --production
